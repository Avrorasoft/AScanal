import os
import shutil
import time

# --- CONFIGURACIÓN ---
CARPETA_DESCARGAS = r"C:\Users\TuUsuario\Downloads" # <--- AJUSTAR ESTO
CARPETA_DESTINO = os.path.join(os.getcwd(), "Entregas_Gemini") 
PREFIJO_PROYECTO = "tva_" 
EXTENSIONES_PERMITIDAS = ['.txt', '.dart', '.json', '.py']

def mover_archivos():
    if not os.path.exists(CARPETA_DESTINO):
        os.makedirs(CARPETA_DESTINO)
    
    for archivo in os.listdir(CARPETA_DESCARGAS):
        ruta_completa_origen = os.path.join(CARPETA_DESCARGAS, archivo)
        if os.path.isfile(ruta_completa_origen) and archivo.startswith(PREFIJO_PROYECTO) and any(archivo.endswith(ext) for ext in EXTENSIONES_PERMITIDAS):
            if archivo.endswith('.crdownload') or archivo.endswith('.part'):
                continue
            ruta_completa_destino = os.path.join(CARPETA_DESTINO, archivo)
            try:
                shutil.move(ruta_completa_origen, ruta_completa_destino)
                print(f"✅ Interceptado: {archivo} -> Entregas_Gemini")
            except Exception as e:
                print(f"⚠️ Error: {e}")

print(f"👁️‍🗨️ VIGILANTE AVRORA ACTIVO.")
try:
    while True:
        mover_archivos()
        time.sleep(2)
except KeyboardInterrupt:
    print("\n🛑 Desactivado.")
