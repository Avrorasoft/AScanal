class _RootControllerState extends State<RootController> {
  final String streamUrl = "https://envivo.canal9.top/live/tv/index.m3u8";
  final String feedJsonUrl = "https://vod.canal9.top/noticias.json";
  final String accessControlUrl = "https://vod.canal9.top/access_control.json";
  static const _linkChannel = MethodChannel('com.avrorasoft.tva/deeplink');

  String r2AccessKey = "";
  String r2SecretKey = "";

  final double baseLat = -11.0065;
  final double baseLng = -66.0631;
  double radioKm = 150.0;
  bool salidaGlobal = false;
  String idHardware = "";
  List<dynamic> materialesFeed = [];
  bool analizando = true;
  String logSeguridad = "";
  bool zonaPermitida = true;
  
  // Añadidas las variables para recibir el evento global desde el backend
  String? tituloEvento;
  String? fechaEventoIso;

  @override
  void initState() {
    super.initState();
    _suscribirEventosPush();
    WidgetsBinding.instance.addPostFrameCallback(
      (_) => _ejecutarProtocoloMaestro(),
    );
    _configurarOyenteDeepLinks();
  }

  void _suscribirEventosPush() async {
    await FirebaseMessaging.instance.subscribeToTopic('tva_noticias');
    FirebaseMessaging.onMessage.listen((RemoteMessage message) {
      if (message.data['accion'] == 'forzar_actualizacion') {
        _sincronizarFeedSilencioso();
      } else if (message.data['accion'] == 'forzar_reinicio_seguridad') {
        _ejecutarProtocoloMaestro();
      }
    });
  }

  Future<void> _sincronizarFeedSilencioso() async {
    if (feedJsonUrl.isEmpty) return;
    try {
      final resFeed = await http.get(
        Uri.parse("$feedJsonUrl?t=${DateTime.now().millisecondsSinceEpoch}"),
      );
      if (resFeed.statusCode == 200) {
        final nuevosMateriales =
            json.decode(resFeed.body)['feed_noticias'] ?? [];
        if (mounted) {
          setState(() {
            materialesFeed = nuevosMateriales;
          });
        }
      }
    } catch (e) {
      debugPrint("Error en sincronización reactiva: $e");
    }
  }

  void _configurarOyenteDeepLinks() async {
    try {
      final String? linkInicial =
          await _linkChannel.invokeMethod('getInitialLink');
      if (linkInicial != null) {
        _procesarUrlDeepLink(linkInicial);
      }
    } catch (e) {
      debugPrint("Error al recibir enlace inicial: $e");
    }

    _linkChannel.setMethodCallHandler((call) async {
      if (call.method == "onLinkReceived") {
        final String? link = call.arguments as String?;
        if (link != null) {
          _procesarUrlDeepLink(link);
        }
      }
    });
  }

  void _procesarUrlDeepLink(String url) {
    try {
      final uri = Uri.parse(url);
      if (uri.pathSegments.contains('noticia')) {
        final idx = uri.pathSegments.indexOf('noticia');
        if (idx + 1 < uri.pathSegments.length) {
          final idNoticia = uri.pathSegments[idx + 1];
          _redireccionarANoticiaDirecta(idNoticia);
        }
      }
    } catch (e) {
      debugPrint("Error al procesar URL: $e");
    }
  }

  void _redireccionarANoticiaDirecta(String idNoticia) async {
    final snapshot =
        await FirebaseDatabase.instance.ref('noticias/$idNoticia').get();

    if (snapshot.exists) {
      final data = snapshot.value as Map<dynamic, dynamic>;
      final urlVideo = data['videoUrl']?.toString() ?? "";

      if (urlVideo.isNotEmpty) {
        FirebaseDatabase.instance
            .ref('noticias/$idNoticia/vistos')
            .runTransaction((Object? v) {
          if (v == null) return Transaction.success(1);
          return Transaction.success((v as int) + 1);
        });
        activeMediaNotifier.value = urlVideo;
        llavedeNavegacionGlobal.currentState?.push(
          MaterialPageRoute(
            builder: (c) => ReproductorVODForzado(urlVideo: urlVideo),
          ),
        );
      }
    }
  }

  Future<bool> _mostrarDivulgacionUbicacion() async {
    return await showDialog<bool>(
          context: context,
          barrierDismissible: false,
          builder: (ctx) => AlertDialog(
            backgroundColor: colorTarjetaOscura,
            shape: RoundedRectangleBorder(
              borderRadius: BorderRadius.circular(15),
            ),
            title: const Row(
              children: [
                Icon(Icons.location_on_rounded, color: colorVerdeTva), // Verde
                SizedBox(width: 10),
                Expanded(
                  child: Text(
                    "Uso de Ubicación",
                    style: TextStyle(
                      color: Colors.white,
                      fontWeight: FontWeight.bold,
                      fontSize: 16,
                    ),
                  ),
                ),
              ],
            ),
            content: const Text(
              "Canal 9 tva recopila datos de ubicación para habilitar la transmisión en vivo únicamente cuando la aplicación está en uso. Esto es necesario para garantizar que el contenido se visualice exclusivamente dentro de las zonas geográficas autorizadas por derechos de transmisión.",
              style: TextStyle(fontSize: 13, color: Colors.white70),
            ),
            actions: [
              TextButton(
                onPressed: () => Navigator.pop(ctx, false),
                child: const Text(
                  "DENEGAR",
                  style: TextStyle(
                    color: Colors.white54,
                    fontWeight: FontWeight.bold,
                  ),
                ),
              ),
              ElevatedButton(
                style: ElevatedButton.styleFrom(
                    backgroundColor: colorVerdeTva), // Verde
                onPressed: () => Navigator.pop(ctx, true),
                child: const Text(
                  "ACEPTAR",
                  style: TextStyle(
                    color: Colors.white,
                    fontWeight: FontWeight.bold,
                  ),
                ),
              ),
            ],
          ),
        ) ??
        false;
  }

  Future<void> _ejecutarProtocoloMaestro() async {
    setState(() {
      analizando = true;
      logSeguridad = "";
      zonaPermitida = true;
      tituloEvento = null;
      fechaEventoIso = null;
    });

    try {
      final prefs = await SharedPreferences.getInstance();
      idHardware = prefs.getString('app_instance_id') ?? "";
      if (idHardware.isEmpty) {
        idHardware =
            "app_${DateTime.now().millisecondsSinceEpoch}_${Random().nextInt(99999)}";
        await prefs.setString('app_instance_id', idHardware);
      }

      try {
        final secureSnapshot = await FirebaseDatabase.instance
            .ref('secure_config')
            .get()
            .timeout(const Duration(seconds: 10));
        if (secureSnapshot.exists) {
          final secureData = secureSnapshot.value as Map<dynamic, dynamic>;
          r2AccessKey = secureData['r2_access_key'] ?? "";
          r2SecretKey = secureData['r2_secret_key'] ?? "";
        }
      } catch (e) {
        debugPrint("Fallo al obtener llaves: $e");
      }

      final resAccess = await http
          .get(Uri.parse(
              "$accessControlUrl?t=${DateTime.now().millisecondsSinceEpoch}"))
          .timeout(const Duration(seconds: 10));

      if (resAccess.statusCode == 200) {
        final accessData = json.decode(resAccess.body);
        salidaGlobal = accessData['global_open'] ?? false;
        radioKm = (accessData['radio_km'] ?? 150.0).toDouble();
        tituloEvento = accessData['evento_titulo'];
        fechaEventoIso = accessData['evento_fecha'];

        if ((accessData['banned_ids'] ?? []).contains(idHardware)) {
          setState(() {
            logSeguridad = "ACCESO DENEGADO: Dispositivo bloqueado.";
            zonaPermitida = false;
            analizando = false;
          });
          return;
        }
      }

      if (!salidaGlobal) {
        List<NetworkInterface> interfaces = await NetworkInterface.list(
          includeLoopback: false,
          type: InternetAddressType.any,
        );

        if (interfaces.any((i) =>
            i.name.toLowerCase().contains("tun") ||
            i.name.toLowerCase().contains("ppp"))) {
          zonaPermitida = false;
          logSeguridad =
              "ACCESO DENEGADO: Uso de red virtual (VPN) no permitido.";
        } else {
          bool gpsActivo = await Geolocator.isLocationServiceEnabled();
          if (!gpsActivo) {
            zonaPermitida = false;
            logSeguridad = "GPS desactivado.";
          } else {
            LocationPermission perm = await Geolocator.checkPermission();
            if (perm == LocationPermission.denied) {
              bool aceptado = await _mostrarDivulgacionUbicacion();
              if (aceptado) {
                perm = await Geolocator.requestPermission();
              }
            }
            if (perm == LocationPermission.denied ||
                perm == LocationPermission.deniedForever) {
              zonaPermitida = false;
              logSeguridad =
                  "Permiso de ubicación requerido para verificar derechos de transmisión local.";
            } else {
              Position pos = await Geolocator.getCurrentPosition(
                  desiredAccuracy: LocationAccuracy.best);
              if (pos.isMocked) {
                zonaPermitida = false;
                logSeguridad = "ACCESO DENEGADO: Ubicación simulada detectada.";
              } else {
                bool exclusionBrasil =
                    pos.longitude > -65.35 && pos.latitude > -11.0;
                double distKm = Geolocator.distanceBetween(
                      pos.latitude,
                      pos.longitude,
                      baseLat,
                      baseLng,
                    ) /
                    1000;
                if (distKm > radioKm || exclusionBrasil) {
                  zonaPermitida = false;
                  logSeguridad =
                      "RESTRICCIÓN GEOGRÁFICA: Señal fuera de cobertura autorizada.";
                }
              }
            }
          }
        }
      }

      final resFeed = await http.get(
          Uri.parse("$feedJsonUrl?t=${DateTime.now().millisecondsSinceEpoch}"));
      if (resFeed.statusCode == 200) {
        materialesFeed = json.decode(resFeed.body)['feed_noticias'] ?? [];
      }

      setState(() {
        analizando = false;
      });
    } catch (e) {
      setState(() {
        logSeguridad = "ERROR DE SISTEMA: Verifique su conexión.";
        analizando = false;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    if (analizando) return _construirPantallaCarga();

    return PantallaGlobal(
      urlSenal: streamUrl,
      materiales: materialesFeed,
      appInstanceId: idHardware,
      radioKm: radioKm,
      salidaGlobal: salidaGlobal,
      zonaPermitida: zonaPermitida,
      tituloEvento: tituloEvento,
      fechaEventoIso: fechaEventoIso,
      r2AccessKey: r2AccessKey,
      r2SecretKey: r2SecretKey,
      onRefresh: _ejecutarProtocoloMaestro,
    );
  }

  Widget _construirPantallaCarga() {
    return Scaffold(
      backgroundColor: colorFondoElegante,
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Image.asset(
              'assets/logo.png',
              height: 80,
              errorBuilder: (c, e, s) => const CircularProgressIndicator(
                  color: Color.fromARGB(255, 2, 148, 50)), // Verde
            ),
            const SizedBox(height: 35),
            if (logSeguridad.isNotEmpty)
              Padding(
                padding: const EdgeInsets.symmetric(horizontal: 40),
                child: Text(
                  logSeguridad,
                  textAlign: TextAlign.center,
                  style: const TextStyle(
                    color: Colors.white54,
                    fontSize: 13,
                    fontWeight: FontWeight.bold,
                  ),
                ),
              ),
          ],
        ),
      ),
    );
  }
}

class _FranjaClimaHoraState extends State<FranjaClimaHora> {
  String _fechaHora = "";
  int _idxCiudad = 0;
  Timer? _reloj;
  Timer? _carrusel;

  final List<Map<String, dynamic>> _ciudades = [
    {
      "nombre": "Riberalta",
      "lat": -11.0065,
      "lng": -66.0631,
      "temp": "--",
      "code": -1
    },
    {
      "nombre": "Guayaramerín",
      "lat": -10.8256,
      "lng": -65.3553,
      "temp": "--",
      "code": -1
    },
    {
      "nombre": "Cobija",
      "lat": -11.0267,
      "lng": -68.7692,
      "temp": "--",
      "code": -1
    },
    {
      "nombre": "Trinidad",
      "lat": -14.8333,
      "lng": -64.9000,
      "temp": "--",
      "code": -1
    },
    {
      "nombre": "La Paz",
      "lat": -16.5000,
      "lng": -68.1193,
      "temp": "--",
      "code": -1
    },
    {
      "nombre": "Cochabamba",
      "lat": -17.3895,
      "lng": -66.1568,
      "temp": "--",
      "code": -1
    },
    {
      "nombre": "Santa Cruz",
      "lat": -17.8146,
      "lng": -63.1361,
      "temp": "--",
      "code": -1
    },
  ];

  @override
  void initState() {
    super.initState();
    _actualizarReloj();
    _reloj =
        Timer.periodic(const Duration(seconds: 30), (_) => _actualizarReloj());
    _cargarClimas();

    _carrusel = Timer.periodic(const Duration(seconds: 5), (_) {
      if (mounted) {
        setState(() {
          _idxCiudad = (_idxCiudad + 1) % _ciudades.length;
        });
      }
    });
  }

  void _actualizarReloj() {
    final now = DateTime.now();
    List<String> meses = [
      "Ene",
      "Feb",
      "Mar",
      "Abr",
      "May",
      "Jun",
      "Jul",
      "Ago",
      "Sep",
      "Oct",
      "Nov",
      "Dic"
    ];
    if (mounted) {
      setState(() {
        _fechaHora =
            "${now.day} ${meses[now.month - 1]} • ${now.hour.toString().padLeft(2, '0')}:${now.minute.toString().padLeft(2, '0')}";
      });
    }
  }

  Future<void> _cargarClimas() async {
    for (var i = 0; i < _ciudades.length; i++) {
      try {
        final res = await http
            .get(Uri.parse(
                'https://api.open-meteo.com/v1/forecast?latitude=${_ciudades[i]["lat"]}&longitude=${_ciudades[i]["lng"]}&current_weather=true'))
            .timeout(const Duration(seconds: 5));
        if (res.statusCode == 200) {
          final data = json.decode(res.body);
          if (mounted) {
            setState(() {
              _ciudades[i]["temp"] =
                  "${data['current_weather']['temperature'].round()}°C";
              _ciudades[i]["code"] = data['current_weather']['weathercode'];
            });
          }
        }
      } catch (e) {}
    }
  }

  IconData _getIconoClima(int code) {
    if (code == -1) return Icons.cloud_off_rounded;
    if (code == 0) return Icons.wb_sunny_rounded;
    if (code >= 1 && code <= 3) return Icons.wb_cloudy_rounded;
    if (code >= 45 && code <= 48) return Icons.foggy;
    if ((code >= 51 && code <= 67) || (code >= 80 && code <= 82))
      return Icons.water_drop_rounded;
    if (code >= 71 && code <= 77) return Icons.ac_unit_rounded;
    if (code >= 95 && code <= 99) return Icons.flash_on_rounded;
    return Icons.cloud_rounded;
  }

  @override
  void dispose() {
    _reloj?.cancel();
    _carrusel?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final int codeActual = _ciudades[_idxCiudad]['code'];
    final Color iconColor = codeActual == 0
        ? Colors.yellowAccent
        : (codeActual >= 51 && codeActual <= 67
            ? Colors.lightBlueAccent
            : Colors.white70);

    return Container(
      color: colorVerdeTva, // FRANJA EN VERDE AMAZONÍA
      padding: const EdgeInsets.symmetric(horizontal: 10),
      child: Row(
        mainAxisAlignment: MainAxisAlignment.spaceBetween,
        children: [
          Text(
            _fechaHora,
            style: const TextStyle(
              color: Colors.white,
              fontSize: 13,
              fontWeight: FontWeight.bold,
              letterSpacing: 1.0,
            ),
          ),
          Row(
            children: [
              Icon(_getIconoClima(codeActual), color: iconColor, size: 16),
              const SizedBox(width: 6),
              AnimatedSwitcher(
                duration: const Duration(milliseconds: 500),
                child: Text(
                  "${_ciudades[_idxCiudad]['nombre']} ${_ciudades[_idxCiudad]['temp']}",
                  key: ValueKey<int>(_idxCiudad),
                  style: const TextStyle(
                    color: Colors.white,
                    fontSize: 12,
                    fontWeight: FontWeight.bold,
                  ),
                ),
              ),
            ],
          ),
        ],
      ),
    );
  }
}

class EscanerCastingRadar extends StatefulWidget {
  const EscanerCastingRadar({super.key});

  @override

class _EscanerCastingRadarState extends State<EscanerCastingRadar> {
  static const _castPlatform = MethodChannel('com.avrorasoft.tva/cast');
  bool _buscando = true;
  final List<String> _dispositivosEncontrados = [];

  @override
  void initState() {
    super.initState();
    _iniciarEscaneoRed();
  }

  Future<void> _iniciarEscaneoRed() async {
    final MDnsClient client = MDnsClient();
    try {
      await client.start();

      await for (final PtrResourceRecord ptr
          in client.lookup<PtrResourceRecord>(
        ResourceRecordQuery.serverPointer('_googlecast._tcp.local'),
      )) {
        await for (final SrvResourceRecord srv
            in client.lookup<SrvResourceRecord>(
          ResourceRecordQuery.service(ptr.domainName),
        )) {
          if (mounted && !_dispositivosEncontrados.contains(srv.target)) {
            setState(() {
              _dispositivosEncontrados.add(
                  srv.target.replaceAll('.local', ' (Chromecast/Android TV)'));
            });
          }
        }
      }

      await for (final PtrResourceRecord ptr
          in client.lookup<PtrResourceRecord>(
        ResourceRecordQuery.serverPointer('_airplay._tcp.local'),
      )) {
        await for (final SrvResourceRecord srv
            in client.lookup<SrvResourceRecord>(
          ResourceRecordQuery.service(ptr.domainName),
        )) {
          if (mounted && !_dispositivosEncontrados.contains(srv.target)) {
            setState(() {
              _dispositivosEncontrados
                  .add(srv.target.replaceAll('.local', ' (AirPlay/Smart TV)'));
            });
          }
        }
      }
    } catch (e) {
      debugPrint("Error en mDNS: $e");
    } finally {
      client.stop();
      if (mounted) {
        setState(() {
          _buscando = false;
        });
      }
    }
  }

      debugPrint("Fallo al invocar casting nativo: $e");

  }

class _PantallaGlobalState extends State<PantallaGlobal>
    with WidgetsBindingObserver {
  final ScrollController _scrollMaster = ScrollController();
  late VideoPlayerController _videoLiveController;
  Timer? _watchdogTimer;

  bool _liveInit = false;
  bool _liveError = false;
  bool _playerAnclado = false;
  bool _inmersionActiva = false;
  bool _mostrarRetorno = false;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    if (widget.zonaPermitida) {
      _iniciarLiveSoberano();
    }

    _scrollMaster.addListener(() {
      if (!_playerAnclado && _scrollMaster.offset > 350) {
        if (!_mostrarRetorno) setState(() => _mostrarRetorno = true);
      } else {
        if (_mostrarRetorno) setState(() => _mostrarRetorno = false);
      }
    });

    activeMediaNotifier.addListener(_onGlobalMediaChanged);
  }

  void _onGlobalMediaChanged() {
    if (activeMediaNotifier.value.isNotEmpty &&
        activeMediaNotifier.value != "LIVE_TV") {
      if (_videoLiveController.value.isPlaying) {
        _videoLiveController.pause();
      }
    }
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (!widget.zonaPermitida) return;
    if (state == AppLifecycleState.resumed) {
      if (_liveInit &&
          !_liveError &&
          (_playerAnclado || _scrollMaster.offset < 50)) {
        if (activeMediaNotifier.value.isEmpty ||
            activeMediaNotifier.value == "LIVE_TV") {
          _videoLiveController.play();
          activeMediaNotifier.value = "LIVE_TV";
        }
      }
    } else if (state == AppLifecycleState.paused) {
      if (_liveInit) _videoLiveController.pause();
    }
  }

  void _iniciarLiveSoberano() {
    setState(() {
      _liveError = false;
    });

    String urlSegura = widget.urlSenal.replaceAll('http://', 'https://');

    _videoLiveController = VideoPlayerController.networkUrl(
      Uri.parse(urlSegura),
      videoPlayerOptions: VideoPlayerOptions(
        allowBackgroundPlayback: false,
        mixWithOthers: true,
      ),
    )..initialize().then((_) {
        ServicioNormalizadorAudio().normalizarVolumen(_videoLiveController);

        if (mounted) {
          setState(() {
            _liveInit = true;
          });
          if (activeMediaNotifier.value.isEmpty ||
              activeMediaNotifier.value == "LIVE_TV") {
            _videoLiveController.play();
            activeMediaNotifier.value = "LIVE_TV";
          }
        }
      }).catchError((error) {
        if (mounted)
          setState(() {
            _liveError = true;
          });
      });

    _videoLiveController.addListener(() {
      if (_videoLiveController.value.hasError && !_liveError) {
        if (mounted)
          setState(() {
            _liveError = true;
          });
        _reiniciarStream();
      }

      if (_videoLiveController.value.isBuffering) {
        if (_watchdogTimer == null || !_watchdogTimer!.isActive) {
          _watchdogTimer = Timer(const Duration(seconds: 6), () {
            if (mounted && _videoLiveController.value.isBuffering) {
              _reiniciarStream();
            }
          });
        }
      } else {
        _watchdogTimer?.cancel();
      }
    });
  }

  void _reiniciarStream() {
    _watchdogTimer?.cancel();
    _videoLiveController.dispose();
    _liveInit = false;
    Future.delayed(const Duration(seconds: 2), () {
      if (mounted && widget.zonaPermitida) _iniciarLiveSoberano();
    });
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    activeMediaNotifier.removeListener(_onGlobalMediaChanged);
    _watchdogTimer?.cancel();
    _scrollMaster.dispose();
    if (widget.zonaPermitida) _videoLiveController.dispose();
    super.dispose();
  }

  void _irAMaterialInmersivo(String urlVideo) async {
    activeMediaNotifier.value = urlVideo;
    if (widget.zonaPermitida && _liveInit) _videoLiveController.pause();

    await Navigator.push(
      context,
      MaterialPageRoute(
        builder: (c) => ReproductorVODForzado(urlVideo: urlVideo),

class _SliverPlayerDelegate extends SliverPersistentHeaderDelegate {
  final double max;
  final Widget child;

  _SliverPlayerDelegate({required this.max, required this.child});

  @override
  double get minExtent => max;

  @override
  double get maxExtent => max;

  @override
  Widget build(
      BuildContext context, double shrinkOffset, bool overlapsContent) {
    return SizedBox.expand(child: child);
  }

  @override
  bool shouldRebuild(covariant _SliverPlayerDelegate oldDelegate) =>
      max != oldDelegate.max || child != oldDelegate.child;
}

            setState(() {
              _eventoActivo = true;
              _tiempoRestante = "TRANSMISIÓN INICIADA";
            });
          }
        } else {
          String dias = diferencia.inDays > 0 ? "${diferencia.inDays}d " : "";

          String minutos = (diferencia.inMinutes % 60).toString().padLeft(2, '0');

  void _conectarDatosEnTiempoReal() {
    if (widget.data['id'] == null) return;

    FirebaseDatabase.instance
        .ref('noticias/${widget.data['id']}/bueno')
        .onValue
        .listen((event) {
      if (event.snapshot.value != null && mounted) {
        _likesNotifier.value =
            int.tryParse(event.snapshot.value.toString()) ?? 0;
      }
    });

    FirebaseDatabase.instance
        .ref('noticias/${widget.data['id']}/vistos')
        .onValue
        .listen((event) {
      if (event.snapshot.value != null && mounted) {
        _vistosNotifier.value =
            int.tryParse(event.snapshot.value.toString()) ?? 0;
      }
    });

    FirebaseDatabase.instance
        .ref('comentarios/${widget.data['id']}')
        .onValue
        .listen((event) {
      if (event.snapshot.value != null && mounted) {
        final map = event.snapshot.value as Map<dynamic, dynamic>;
        List<Map<String, dynamic>> temp = [];
        int totalSum = 0;

        map.forEach((key, val) {
          List<Map<String, String>> respuestasTemp = [];
          if (val['respuestas'] != null) {
            (val['respuestas'] as Map<dynamic, dynamic>).forEach((
              rKey,
              rVal,
            ) {
              respuestasTemp.add({
                'key': rKey.toString(),
                'tipo': rVal['tipo']?.toString() ?? 'texto',
                'contenido': rVal['contenido'].toString(),
                'apodo': rVal['apodo']?.toString() ?? 'Anónimo',
                'oficial': rVal['oficial']?.toString() ?? 'false',
              });
            });
          }
          temp.add({
            'key': key.toString(),
            'tipo': val['tipo'].toString(),
            'contenido': val['contenido'].toString(),
            'apodo': val['apodo']?.toString() ?? 'Anónimo',
            'oficial': val['oficial']?.toString() ?? 'false',
            'respuestas': respuestasTemp,
          });
        });

        temp.sort((a, b) {
          if (a['oficial'] == 'true' && b['oficial'] == 'false') return -1;
          if (a['oficial'] == 'false' && b['oficial'] == 'true') return 1;
          return 0;
        });

        for (var c in temp) {
          totalSum += 1 + (c['respuestas'] as List).length;
        }

        _comentariosNotifier.value = temp;
        _totalComentariosNotifier.value = totalSum;
      } else if (mounted) {
        _comentariosNotifier.value = [];
        _totalComentariosNotifier.value = 0;
      }
    });
  }

  Future<void> _accionarAudio(String url) async {
    String urlSegura = url.replaceAll('http://', 'https://');

    if (_audioStateNotifier.value['url'] == urlSegura &&
        _reproductorAudioUsuario != null) {
      if (_reproductorAudioUsuario!.value.isPlaying) {
        _reproductorAudioUsuario!.pause();
        _audioStateNotifier.value = {
          'url': urlSegura,
          'isPlaying': false,
          'progress': _audioStateNotifier.value['progress'],
          'isLoading': false,
        };
      } else {
        activeMediaNotifier.value = urlSegura;
        _reproductorAudioUsuario!.play();
        _audioStateNotifier.value = {
          'url': urlSegura,
          'isPlaying': true,
          'progress': _audioStateNotifier.value['progress'],
          'isLoading': false,
        };
      }
      return;
    }

    activeMediaNotifier.value = urlSegura;
    _audioStateNotifier.value = {
      'url': urlSegura,
      'isPlaying': false,
      'progress': 0.0,
      'isLoading': true,
    };
    _reproductorAudioUsuario?.dispose();
    _reproductorAudioUsuario = VideoPlayerController.networkUrl(
      Uri.parse(urlSegura.trim()),
    );

    try {
      await _reproductorAudioUsuario!.initialize();
      _reproductorAudioUsuario!.setVolume(1.0);
      _reproductorAudioUsuario!.play();

      _reproductorAudioUsuario!.addListener(() {
        if (mounted && _reproductorAudioUsuario != null) {
          double prog = 0.0;
          if (_reproductorAudioUsuario!.value.duration.inMilliseconds > 0) {
            prog = _reproductorAudioUsuario!.value.position.inMilliseconds /
                _reproductorAudioUsuario!.value.duration.inMilliseconds;
          }
          bool playing = _reproductorAudioUsuario!.value.isPlaying;

          if (_reproductorAudioUsuario!.value.position >=
                  _reproductorAudioUsuario!.value.duration &&
              _reproductorAudioUsuario!.value.duration > Duration.zero) {
            _reproductorAudioUsuario!.pause();
            _reproductorAudioUsuario!.seekTo(Duration.zero);
            playing = false;
            prog = 0.0;
          }

          _audioStateNotifier.value = {
            'url': urlSegura,
            'isPlaying': playing,
            'progress': prog,
            'isLoading': false,
          };
        }
      });
    } catch (e) {
      _audioStateNotifier.value = {
        'url': '',
        'isPlaying': false,
        'progress': 0.0,
        'isLoading': false,
      };
    }
  }

  Map<String, String> _firmarPeticionS3(
    String method,
    String url,
    List<int> payload,
  ) {
    final accessKey = widget.r2AccessKey;
    final secretKey = widget.r2SecretKey;

    final uri = Uri.parse(url);
    final host = uri.host;
    final path = uri.path;

    final now = DateTime.now().toUtc();
    final amzDate =
        "${now.year.toString().padLeft(4, '0')}${now.month.toString().padLeft(2, '0')}${now.day.toString().padLeft(2, '0')}T${now.hour.toString().padLeft(2, '0')}${now.minute.toString().padLeft(2, '0')}${now.second.toString().padLeft(2, '0')}Z";
    final dateStamp = amzDate.substring(0, 8);

    final payloadHash = sha256.convert(payload).toString();
    final canonicalHeaders =
        "host:$host\nx-amz-content-sha256:$payloadHash\nx-amz-date:$amzDate\n";
    const signedHeaders = "host;x-amz-content-sha256;x-amz-date";

    final canonicalRequest =
        "$method\n$path\n\n$canonicalHeaders\n$signedHeaders\n$payloadHash";
    final credentialScope = "$dateStamp/auto/s3/aws4_request";
    final stringToSign =
        "AWS4-HMAC-SHA256\n$amzDate\n$credentialScope\n${sha256.convert(utf8.encode(canonicalRequest))}";

    List<int> sign(List<int> key, String msg) =>
        Hmac(sha256, key).convert(utf8.encode(msg)).bytes;

    final kDate = sign(utf8.encode("AWS4$secretKey"), dateStamp);
    final signature = Hmac(
      sha256,
      sign(sign(sign(kDate, "auto"), "s3"), "aws4_request"),
    ).convert(utf8.encode(stringToSign)).toString();

    return {
      'x-amz-date': amzDate,
      'x-amz-content-sha256': payloadHash,
      'Authorization':
          "AWS4-HMAC-SHA256 Credential=$accessKey/$credentialScope, SignedHeaders=$signedHeaders, Signature=$signature",
      'Content-Type': 'audio/x-m4a',
    };
  }

  Future<void> _subirAudioAR2(String path, [String? parentId]) async {
    String fileName = "audio_${DateTime.now().millisecondsSinceEpoch}.m4a";

    try {
      final file = File(path);
      final bytes = await file.readAsBytes();
      final url = "$r2Endpoint/$fileName";

      var response = await http.put(
        Uri.parse(url),
        headers: _firmarPeticionS3('PUT', url, bytes),
        body: bytes,
      );

      if ((response.statusCode == 200 || response.statusCode == 201) &&
          widget.data['id'] != null) {
        String ruta = parentId == null
            ? 'comentarios/${widget.data['id']}'
            : 'comentarios/${widget.data['id']}/$parentId/respuestas';
        FirebaseDatabase.instance.ref(ruta).push().set({
          'tipo': 'audio',
          'contenido': "$r2PublicUrl/$fileName",
          'apodo': _miApodo ?? 'Anónimo',
          'oficial': 'false',
          'fecha': DateTime.now().toIso8601String(),
        });
      }
    } catch (e) {
      debugPrint("Error R2: $e");
    }
  }

  void _gestionarGrabacionUnificada(String? parentId, StateSetter setBtnState) async {

      } catch (e) {
        setState(() => _grabandoComentario = false);
        setBtnState(() => _grabandoComentario = false);
      }
    } else {
      _detenerAudio();
      try {
        if (await Permission.microphone.request().isGranted) {
          final dir = await getTemporaryDirectory();
          String path = '${dir.path}/audio_${DateTime.now().millisecondsSinceEpoch}.m4a';
          if (await _audioPlatform.invokeMethod('startRecording', {"path": path, "noiseSuppression": true})) {
            HapticFeedback.heavyImpact();
            setState(() { _grabandoComentario = true; _idComentarioActual = parentId; });
            setBtnState(() => _grabandoComentario = true);
            ScaffoldMessenger.of(context).showSnackBar(
              SnackBar(
                content: const Text("🔴 Grabando..."),
                backgroundColor: Colors.red,

                action: SnackBarAction(label: "ENVIAR", textColor: Colors.white, onPressed: () => _gestionarGrabacionUnificada(_idComentarioActual, setBtnState)),

      }
    }

            const Text("¿Cómo deseas responder?", style: TextStyle(color: Colors.white, fontWeight: FontWeight.w900, fontSize: 16)),
            const SizedBox(height: 20),
            Row(
              mainAxisAlignment: MainAxisAlignment.spaceEvenly,
              children: [
                _BotonOpcionRespuesta(
                  icono: Icons.keyboard_alt_outlined,
                  texto: "Escrito",
                  color: Colors.white,
                  onTap: () {
                    Navigator.pop(ctx);
                    _mostrarDialogoTexto(parentId, responderA);
                  },
                ),
                _BotonOpcionRespuesta(
                  icono: Icons.mic_none_rounded,
                  texto: "Voz",
                  color: colorRojoTva,
                  onTap: () {
                    Navigator.pop(ctx);
                    if (_setModalMicState != null) {
                      _gestionarGrabacionUnificada(parentId, _setModalMicState!);
                    }
                  },
                ),
              ],
            ),
          ],
        ),
      ),
    ); 
  }

  void _mostrarDialogoTexto(String? parentId, [String? responderA]) {
    _controladorTextoComentario.clear();
    if (responderA != null) _controladorTextoComentario.text = "@$responderA ";
    bool marcarOficial = false;

    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      useSafeArea: true, 
      builder: (context) => StatefulBuilder(
        builder: (context, setModalState) {
          return Padding(
            padding: EdgeInsets.only(bottom: MediaQuery.of(context).viewInsets.bottom),
            child: Container(
              decoration: const BoxDecoration(
                color: colorTarjetaOscura,
                borderRadius: BorderRadius.vertical(top: Radius.circular(15)),
              ),
              padding: const EdgeInsets.all(20),
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Text(parentId == null ? "Escribir Comentario" : "Responder", style: const TextStyle(color: Colors.white, fontSize: 14, fontWeight: FontWeight.bold)),
                  const SizedBox(height: 15),
                  TextField(
                    controller: _controladorTextoComentario,
                    style: const TextStyle(color: Colors.white),
                    maxLines: 4,
                    decoration: const InputDecoration(border: OutlineInputBorder(), hintText: "Escriba aquí...", hintStyle: TextStyle(color: Colors.white38), isDense: true),
                  ),
                  const SizedBox(height: 10),
                  if (_isModoDios && parentId == null)
                    CheckboxListTile(
                      title: const Text("Fijar como Anuncio Oficial", style: TextStyle(color: colorRojoTva, fontWeight: FontWeight.bold, fontSize: 12)),
                      value: marcarOficial,
                      onChanged: (bool? value) { setModalState(() { marcarOficial = value ?? false; }); },
                      controlAffinity: ListTileControlAffinity.leading,
                      activeColor: colorRojoTva,
                    ),
                  const SizedBox(height: 10),
                  Row(
                    mainAxisAlignment: MainAxisAlignment.end,
                    children: [
                      TextButton(onPressed: () => Navigator.pop(context), child: const Text("CANCELAR", style: TextStyle(color: Colors.white54, fontSize: 12))),
                      ElevatedButton(
                        style: ElevatedButton.styleFrom(backgroundColor: colorRojoTva),
                        onPressed: () {
                          if (_controladorTextoComentario.text.trim().isNotEmpty && widget.data['id'] != null) {
                            String ruta = parentId == null ? 'comentarios/${widget.data['id']}' : 'comentarios/${widget.data['id']}/${parentId}/respuestas';
                            FirebaseDatabase.instance.ref(ruta).push().set({
                              'tipo': 'texto',
                              'contenido': _controladorTextoComentario.text.trim(),
                              'apodo': _miApodo ?? 'Anónimo',
                              'oficial': marcarOficial.toString(),
                              'fecha': DateTime.now().toIso8601String(),
                            });
                          }
                          Navigator.pop(context);
                        },
                        child: const Text("ENVIAR", style: TextStyle(color: Colors.white, fontSize: 12)),
                      ),
                    ],
                  ),
                ],
              ),
            ),
          );
        },
      ),
    ); 
  }

  void _compartirNoticia() {
    String link = widget.data['id'] != null
        ? "https://vod.canal9.top/noticia/${widget.data['id']}"
        : "https://play.google.com/store/apps/details?id=com.avrorasoft.tva";
    String textoACompartir =
        "🔴 NOTICIA - CANAL 9 tva\n${widget.data['titulo'] ?? ''}\n\n📱 Mira el reporte completo y nuestra señal en vivo descargando la app oficial:\n$link";
    Share.share(textoACompartir);
  }

  Widget _iconAction(
    IconData icon,
    String value,
    VoidCallback? onTap, {
    Color color = Colors.white70,
  }) {
    return GestureDetector(
      onTap: onTap,
      behavior: HitTestBehavior.opaque,
      child: Padding(
        padding: const EdgeInsets.symmetric(vertical: 20.0, horizontal: 15.0),
        child: Row(
          mainAxisSize: MainAxisSize.min,
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Icon(icon, size: 28, color: color),
            if (value.isNotEmpty) const SizedBox(width: 6),
            if (value.isNotEmpty)
              Text(value, style: TextStyle(fontSize: 14, fontWeight: FontWeight.bold, color: color)),
          ],
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    String urlVideo = widget.data['videoUrl']?.toString().trim() ?? "";
    bool esVideo = urlVideo.isNotEmpty &&
        (urlVideo.endsWith('.mp4') || urlVideo.endsWith('.m3u8'));

    String urlImg = widget.data['imagen']?.toString().trim() ?? "";
    if (urlImg.isEmpty && widget.data['imagenes'] != null) {
      List<String> imgs = List<String>.from(widget.data['imagenes']);
      if (imgs.isNotEmpty) urlImg = imgs[0];
    }

    urlImg = urlImg.replaceAll('http://', 'https://');

    return Column(
      children: [
        Container(height: 6, width: double.infinity, color: colorFondoElegante),
        Container(
          color: colorTarjetaOscura,
          padding: const EdgeInsets.only(top: 15, bottom: 5),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Padding(
                padding: const EdgeInsets.symmetric(horizontal: 15),
                child: Row(
                  mainAxisAlignment: MainAxisAlignment.spaceBetween,
                  children: [
                    const Text("INFORMACIÓN", style: TextStyle(fontSize: 10, fontWeight: FontWeight.w900, color: colorRojoTva)),
                    Text(widget.data['fecha']?.toUpperCase() ?? "", style: const TextStyle(fontSize: 10, fontWeight: FontWeight.w900, color: Colors.white54)),
                  ],
                ),
              ),
              const SizedBox(height: 8),
              Padding(
                padding: const EdgeInsets.symmetric(horizontal: 15),
                child: Text(widget.data['titulo'] ?? "", style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 13, color: Colors.white, height: 1.2)),
              ),
              const SizedBox(height: 12),
              GestureDetector(
                onTap: widget.onTap,
                behavior: HitTestBehavior.opaque,
                child: Container(
                  width: double.infinity,
                  height: MediaQuery.of(context).size.height * 0.52,
                  color: Colors.black,
                  child: Stack(
                    alignment: Alignment.center,
                    children: [
                      if (urlImg.isNotEmpty)
                        Image.network(urlImg, fit: BoxFit.cover, width: double.infinity, height: double.infinity, errorBuilder: (_, __, ___) => const Center(child: Icon(Icons.image_not_supported, color: Colors.white24)))
                      else
                        Container(color: const Color(0xFF181818), child: Center(child: Image.asset('assets/logo.png', height: 80, color: Colors.white24))),
                      if (esVideo)
                        const Center(child: Icon(Icons.play_circle_fill_rounded, color: Colors.white70, size: 60)),
                    ],
                  ),
                ),
              ),
              Container(
                color: colorTarjetaOscura,
                padding: const EdgeInsets.symmetric(vertical: 12),
                child: Row(
                  mainAxisAlignment: MainAxisAlignment.spaceEvenly,
                  children: [
                    ValueListenableBuilder<int>(
                      valueListenable: _likesNotifier,
                      builder: (context, likes, _) => _iconAction(_isLiked ? Icons.favorite_rounded : Icons.favorite_border, likes.toString(), _darMeGusta, color: _isLiked ? colorRojoTva : Colors.white70),
                    ),
                    ValueListenableBuilder<int>(
                      valueListenable: _vistosNotifier,
                      builder: (context, vistos, _) => _iconAction(Icons.visibility_outlined, vistos.toString(), null),
                    ),
                    ValueListenableBuilder<int>(
                      valueListenable: _totalComentariosNotifier,
                      builder: (context, comentarios, _) => _iconAction(Icons.chat_bubble_outline, comentarios.toString(), _abrirModalComentarios),
                    ),
                    _iconAction(Icons.share_outlined, "", _compartirNoticia),
                  ],
                ),
              ),
            ],
          ),
        ),
      ],
    );
  }
}

class _BotonOpcionRespuesta extends StatelessWidget {
  final IconData icono;
  final String texto;
  final Color color;
  final VoidCallback onTap;

  const _BotonOpcionRespuesta({
    super.key,
    required this.icono,
    required this.texto,
    required this.color,
    required this.onTap,
  });

  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      onTap: onTap,
      behavior: HitTestBehavior.opaque,
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Container(
            padding: const EdgeInsets.all(15),
            decoration: BoxDecoration(color: color.withOpacity(0.1), shape: BoxShape.circle),
            child: Icon(icono, color: color, size: 30),
          ),
          const SizedBox(height: 8),
          Text(texto, style: TextStyle(color: color, fontWeight: FontWeight.bold, fontSize: 13)),
        ],
      ),
    );
  }
}
