final ValueNotifier<String> globalAudioNotifier = ValueNotifier<String>('none');

// Control Global de VDO para corte inmediato al desplazar o cambiar de player
WebViewController? globalVdoController;
bool isGlobalVdoInitialized = false;

void forceKillGlobalVdo() {
  if (globalVdoController != null) {
    globalVdoController!.loadRequest(Uri.parse('about:blank'));
    globalVdoController = null;
    isGlobalVdoInitialized = false;
    globalAudioNotifier.value = 'none';
  }
}

class VideoRegistry {
  static final Map<String, GlobalKey> _keys = {};
  
  static GlobalKey getKey(String id) {
    _keys.putIfAbsent(id, () => GlobalKey());
    return _keys[id]!;
  }
}

// Lógica de georrestricción - Sistema métrico
double _calcularDistanciaCanal9(double userLat, double userLon) {
  double ribLat = -10.998041;
  double ribLon = -66.064368;
  var p = 0.017453292519943295;
  var c = cos;
  var a = 0.5 - 
          c((ribLat - userLat) * p) / 2 + 
          c(userLat * p) * c(ribLat * p) * (1 - c((ribLon - userLon) * p)) / 2;
          
  return 12742 * asin(sqrt(a)); // Resultado estricto en kilómetros
}

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> with AutomaticKeepAliveClientMixin {
  final String canalAId = 'UCsFHCdqQP6zwoh8oaloiRHQ'; 
  final String apiKey = 'AIzaSyBNrxQy0-y6GFV17i9vU2MyE89FyHRqDMY';
  
  List<Map<String, dynamic>> _combinedData = [];
  bool _isLoading = true;

  @override
  bool get wantKeepAlive => true;

  @override
  void initState() { 
    super.initState(); 
    _refreshData(); 
  }

  Future<void> _refreshData() async {
    List<Map<String, dynamic>> temp = [];
    
    // 1. Extracción de YouTube Canal A
    try {
      final res = await http.get(Uri.parse(
        'https://www.googleapis.com/youtube/v3/playlistItems?part=snippet&playlistId=UU${canalAId.substring(2)}&maxResults=8&key=$apiKey'
      ));
      
      if (res.statusCode == 200) {
        final decodedData = json.decode(res.body);
        for (var item in decodedData['items']) {
          temp.add({
            'tipo': 'video', 
            'id': item['snippet']['resourceId']['videoId'], 
            'titulo': item['snippet']['title'], 
            'fecha': item['snippet']['publishedAt']
          });
        }
      }
    } catch (_) {}

    // 2. Extracción de RSS Diversificado (60/20/10/10) con RT y Scraping
    temp.addAll(await _fetchDeepCategory('https://eju.tv/feed/', 6)); // 60% Bolivia
    temp.addAll(await _fetchDeepCategory('https://actualidad.rt.com/feeds/all.rss', 1)); // 10% RT Internacional
    temp.addAll(await _fetchDeepCategory('https://www.ngenespanol.com/feed/', 2)); // 20% Curiosidades
    temp.addAll(await _fetchDeepCategory('https://as.com/rss/futbol/primera.xml', 1)); // 10% Deportes

    // Ordenamiento por fecha
    temp.sort((a, b) => b['fecha'].toString().compareTo(a['fecha'].toString()));
    
    if (mounted) {
      setState(() { 
        _combinedData = temp; 
        _isLoading = false; 
      });
    }
  }

  Future<List<Map<String, dynamic>>> _fetchDeepCategory(String url, int limit) async {
    List<Map<String, dynamic>> list = [];
    
    try {
      final res = await http.get(Uri.parse(url)).timeout(const Duration(seconds: 5));
      
      if (res.statusCode == 200) {
        final items = RegExp(r'<item>(.*?)</item>', dotAll: true).allMatches(res.body);
        
        for (var m in items) {
          if (list.length >= limit) break;
          
          final content = m.group(1)!;
          
          final titleMatch = RegExp(r'<title>(.*?)</title>').firstMatch(content);
          final title = titleMatch?.group(1)?.replaceAll('<![CDATA[', '').replaceAll(']]>', '') ?? 'Noticia';
          
          final linkMatch = RegExp(r'<link>(.*?)</link>').firstMatch(content);
          final link = linkMatch?.group(1) ?? '';
          
          final imgMatch = RegExp(r'url="(.*?)"').firstMatch(content);
          final img = imgMatch?.group(1) ?? 'https://images.unsplash.com/photo-1526304640581-d334cdbbf45e?w=600';
          
          String deepText = "Cargando contenido completo...";
          
          try {
            final pRes = await http.get(Uri.parse(link)).timeout(const Duration(seconds: 4));
            if (pRes.statusCode == 200) {
              final ps = RegExp(r'<p>(.*?)</p>', dotAll: true).allMatches(pRes.body);
              deepText = ps.map((e) => e.group(1)!
                  .replaceAll(RegExp(r'<[^>]*>'), '')
                  .trim())
                  .where((s) => s.length > 50)
                  .take(12)
                  .join('\n\n');
            }
          } catch (_) {}

          if (deepText.length < 100) {
            final descMatch = RegExp(r'<description>(.*?)</description>', dotAll: true).firstMatch(content);
            deepText = descMatch?.group(1)?.replaceAll(RegExp(r'<[^>]*>'), '').trim() ?? deepText;
          }

          String resumen = deepText.length > 130 ? '${deepText.substring(0, 130)}...' : deepText;

          list.add({
            'tipo': 'texto', 
            'titulo': title, 
            'cuerpo': deepText, 
            'resumen': resumen, 
            'foto': img, 
            'fecha': DateTime.now().toIso8601String()
          });
        }
      }
    } catch (_) {}
    
    return list;
  }

  @override
  Widget build(BuildContext context) {
    super.build(context);
    
    if (_isLoading) {
      return const Center(
        child: CircularProgressIndicator(color: Color(0xFF00c853)),
      );
    }

    return NotificationListener<ScrollNotification>(
      onNotification: (ScrollNotification scroll) {
        // ===============================================
        // SENSOR 100PX: Apaga el audio de VDO al bajar
        // ===============================================
        if (scroll.metrics.pixels > 100 && globalAudioNotifier.value == 'vdo') {
          forceKillGlobalVdo();
        }
        return false;
      },
      child: StreamBuilder<QuerySnapshot>(
        stream: FirebaseFirestore.instance.collection('noticias').snapshots(),
        builder: (context, snapshot) {
          List<Map<String, dynamic>> fullList = List.from(_combinedData);
          
          if (snapshot.hasData) {
            for (var doc in snapshot.data!.docs) {
              final d = doc.data() as Map<String, dynamic>;
              if (d['esVisible'] == true) {
                fullList.add(d);
              }
            }
          }
          
          fullList.sort((a, b) {
             DateTime dA = a['fecha'] is Timestamp 
                ? (a['fecha'] as Timestamp).toDate() 
                : DateTime.tryParse(a['fecha'].toString()) ?? DateTime.now();
                
             DateTime dB = b['fecha'] is Timestamp 
                ? (b['fecha'] as Timestamp).toDate() 
                : DateTime.tryParse(b['fecha'].toString()) ?? DateTime.now();
                
             return dB.compareTo(dA);
          });

          return ListView.builder(
            padding: const EdgeInsets.all(16),
            itemCount: fullList.length + 2,
            itemBuilder: (context, index) {
              if (index == 0) return const VdoNinjaWrapper();
              if (index == 1) return const HomeButtons();
              
              final item = fullList[index - 2];
              
              return Padding(
                padding: const EdgeInsets.only(bottom: 16),
                child: item['tipo'] == 'video' 
                  ? YoutubeVideoCard(
                      title: item['titulo'], 
                      videoId: item['id'],
                    ) 
                  : NoticiaTextoCard(
                      noticia: item,
                    ),
              );
            },
          );
        },
      ),
    );
  }
}

// REPRODUCTOR DE YOUTUBE EXTRAÍBLE

class ExtractableYoutubePlayer extends StatefulWidget {
  final String videoId;
  
  const ExtractableYoutubePlayer({
    super.key, 
    required this.videoId,
  });

  @override
  State<ExtractableYoutubePlayer> createState() => _ExtractableYoutubePlayerState();
}

class _ExtractableYoutubePlayerState extends State<ExtractableYoutubePlayer> {
  YoutubePlayerController? _controller;
  bool _isInitialized = false;

  void _init() {
    forceKillGlobalVdo(); // Silencia y mata VDO si inicias un video de YouTube
    
    globalAudioNotifier.value = widget.videoId;
    
    _controller = YoutubePlayerController(
      initialVideoId: widget.videoId, 
      flags: const YoutubePlayerFlags(
        autoPlay: true, 
        mute: false,
      ),
    );
    
    setState(() {
      _isInitialized = true;
    });
  }

  @override
  void dispose() { 
    _controller?.dispose(); 
    super.dispose(); 
  }

  @override
  Widget build(BuildContext context) {
    if (_isInitialized && _controller != null) {
      return YoutubePlayer(controller: _controller!);
    }
    
    return Stack(
      fit: StackFit.expand, 
      children: [
        Image.network(
          'https://img.youtube.com/vi/${widget.videoId}/hqdefault.jpg', 
          fit: BoxFit.cover, 
          errorBuilder: (c,e,s) => Container(color: Colors.grey[900]),
        ),
        Container(color: Colors.black45),
        const Center(
          child: Icon(
            Icons.play_circle_fill, 
            size: 60, 
            color: Colors.red,
          ),
        ),
        InkWell(onTap: _init),
      ],
    );
  }
}

            )
          );

    String url = "https://images.unsplash.com/photo-1526304640581-d334cdbbf45e?w=600";
    if (_img != null) {
      try {
        final res = await CloudinaryPublic('da9syyeb2', 'preset_noticias')
            .uploadFile(CloudinaryFile.fromFile(_img!.path));
        url = res.secureUrl;
      } catch (_) {}
    }
    
    await FirebaseFirestore.instance.collection('noticias').add({
      'tipo': 'texto', 
      'titulo': _t.text, 
      'cuerpo': _c.text, 
      'resumen': _c.text.length > 50 ? '${_c.text.substring(0, 50)}...' : _c.text,
      'foto': url, 
      'fecha': Timestamp.now(), 
      'esVisible': true
    });
    
    if (mounted) Navigator.pop(context);
  }

  void _confirmarBorrado(DocumentSnapshot doc) {
    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        backgroundColor: const Color(0xFF1c1c1c),
        title: const Text("Confirmar", style: TextStyle(color: Colors.white)),
        content: const Text(
          "¿Eliminar este material?", 
          style: TextStyle(color: Colors.white70),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context), 
            child: const Text("Cancelar"),
          ),
          ElevatedButton(
            style: ElevatedButton.styleFrom(backgroundColor: Colors.red),
            onPressed: () async {
              try { 
                await doc.reference.delete(); 
                if (mounted) Navigator.pop(context); 
              } 
              catch (e) { 
                if (mounted) { 
                  Navigator.pop(context); 
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(
                      content: Text("Cambia 'allow delete: if true;' en Firebase"),
                    ),
                  ); 
                } 
              }
            },
            child: const Text("Eliminar", style: TextStyle(color: Colors.white)),
          )
        ],
      )
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text("CMS Canal 9 TVA")),
      body: Padding(
        padding: const EdgeInsets.all(16), 
        child: Column(
          children: [
            TextField(
              controller: _t, 
              decoration: const InputDecoration(labelText: "Título"),
            ),
            TextField(
              controller: _c, 
              decoration: const InputDecoration(labelText: "Cuerpo"), 
              maxLines: 3,
            ),
            ElevatedButton(
              onPressed: _pick, 
              child: const Text("Foto"),
            ),
            _loading 
              ? const CircularProgressIndicator() 
              : ElevatedButton(
                  onPressed: _save, 
                  style: ElevatedButton.styleFrom(
                    backgroundColor: const Color(0xFF00c853),
                  ), 
                  child: const Text("PUBLICAR"),
                ),
            const Divider(),
            Expanded(
              child: StreamBuilder<QuerySnapshot>(
                stream: FirebaseFirestore.instance.collection('noticias').orderBy('fecha', descending: true).snapshots(),
                builder: (context, snapshot) {
                  if (!snapshot.hasData) return const SizedBox.shrink();
                  
                  return ListView.builder(
                    itemCount: snapshot.data!.docs.length,
                    itemBuilder: (context, index) {
                      final doc = snapshot.data!.docs[index];
                      final data = doc.data() as Map<String, dynamic>;
                      
                      return Card(
                        child: ListTile(
                          title: Text(
                            data['titulo'] ?? 'Sin título', 
                            maxLines: 1,
                          ), 
                          trailing: IconButton(
                            icon: const Icon(Icons.delete, color: Colors.red), 
                            onPressed: () => _confirmarBorrado(doc),
                          ),
                        ),
                      );
                    },
                  );
                },
              )
            )
          ],
        ),
      ),
    );

// SECCIÓN CLIMA (SISTEMA MÉTRICO)
// ==========================================
class PantallaClimaNativo extends StatefulWidget {
  const PantallaClimaNativo({super.key});
  @override
  State<PantallaClimaNativo> createState() => _PantallaClimaNativoState();
}

class _PantallaClimaNativoState extends State<PantallaClimaNativo> {
  bool _isLoading = true;
  String _temp = '', _desc = '';
  IconData _icon = Icons.cloud;
  List<dynamic> _forecast = [], _hourly = [];

  @override
  void initState() { 
    super.initState(); 
    _fetchClimaRiberalta(); 
  }

  Future<void> _fetchClimaRiberalta() async {
    try {
      final response = await http.get(Uri.parse(
        'https://api.open-meteo.com/v1/forecast?latitude=-10.9980&longitude=-66.0643&current_weather=true&daily=weathercode,temperature_2m_max,temperature_2m_min&hourly=temperature_2m,weathercode&timezone=America/La_Paz'
      ));
      
      if (response.statusCode == 200) {
        final data = json.decode(response.body);
        List<dynamic> forecastList = [];
        
        for (int i = 0; i < data['daily']['time'].length; i++) {
          forecastList.add({
            'date': data['daily']['time'][i], 
            'code': data['daily']['weathercode'][i], 
            'max': data['daily']['temperature_2m_max'][i], 
            'min': data['daily']['temperature_2m_min'][i]
          });
        }
        
        List<dynamic> hourlyList = [];
        for (int i = 0; i < data['hourly']['time'].length; i++) {
          hourlyList.add({
            'time': data['hourly']['time'][i], 
            'temp': data['hourly']['temperature_2m'][i], 
            'code': data['hourly']['weathercode'][i]
          });
        }
        
        if (mounted) {
          setState(() { 
            _temp = '${data['current_weather']['temperature']}°'; 
            _desc = _getWeatherDesc(data['current_weather']['weathercode']); 
            _icon = _getWeatherIcon(data['current_weather']['weathercode']); 
            _forecast = forecastList; 
            _hourly = hourlyList; 
            _isLoading = false; 
          });
        }
      } else {
        if (mounted) setState(() { _desc = 'Error'; _isLoading = false; });
      }
    } catch (e) {
      if (mounted) setState(() { _desc = 'Error'; _isLoading = false; });
    }
  }

  String _getWeatherDesc(int code) {
    if (code == 0) return 'Cielo Despejado'; 
    if (code <= 3) return 'Nublado'; 
    if (code <= 67) return 'Lluvia'; 
    if (code <= 82) return 'Chubascos'; 
    return 'Tormenta';
  }

  IconData _getWeatherIcon(int code) {
    if (code == 0) return Icons.wb_sunny; 
    if (code <= 3) return Icons.cloud_queue; 
    if (code <= 67) return Icons.water_drop; 
    if (code <= 82) return Icons.umbrella; 
    return Icons.flash_on;
  }

  String _getDayName(String d, int index) {
    if (index == 0) return 'Hoy'; 
    if (index == 1) return 'Mañana';
    
    switch (DateTime.parse(d).weekday) { 
      case 1: return 'Lunes'; 
      case 2: return 'Martes'; 
      case 3: return 'Miércoles'; 
      case 4: return 'Jueves'; 
      case 5: return 'Viernes'; 
      case 6: return 'Sábado'; 
      case 7: return 'Domingo'; 
      default: return ''; 
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Pronóstico Riberalta'), 
        leading: IconButton(
          icon: const Icon(Icons.arrow_back, color: Color(0xFF00c853)), 
          onPressed: () => Navigator.pop(context),
        ),
      ),
      body: _isLoading 
        ? const Center(child: CircularProgressIndicator(color: Color(0xFF00c853))) 
        : Stack(
            clipBehavior: Clip.hardEdge, 
            children: [
              Positioned(
                bottom: 10, 
                left: 0, 
                right: 0, 
                child: Column(
                  mainAxisSize: MainAxisSize.min, 
                  children: [
                    const Text(
                      'ESTADO ACTUAL', 
                      style: TextStyle(color: Colors.white54, fontWeight: FontWeight.bold, fontSize: 11),
                    ), 
                    Row(
                      mainAxisAlignment: MainAxisAlignment.center, 
                      children: [
                        Icon(_icon, size: 40, color: const Color(0xFF00c853)), 
                        const SizedBox(width: 8), 
                        Text(_temp, style: const TextStyle(fontSize: 40, fontWeight: FontWeight.bold, color: Colors.white))
                      ],
                    ), 
                    Text(_desc, style: const TextStyle(fontSize: 14, color: Colors.white70))
                  ],
                ),
              ),
              Positioned(
                top: 10, 
                left: 12, 
                right: 12, 
                child: Container(
                  padding: const EdgeInsets.only(top: 8, left: 8, right: 8, bottom: 4), 
                  decoration: BoxDecoration(
                    color: const Color(0xFF1c1c1c).withValues(alpha: 0.98), 
                    borderRadius: BorderRadius.circular(16), 
                    border: Border.all(color: Colors.white10),
                  ), 
                  child: Column(
                    mainAxisSize: MainAxisSize.min, 
                    crossAxisAlignment: CrossAxisAlignment.start, 
                    children: [
                      const Padding(
                        padding: EdgeInsets.only(bottom: 2, left: 8), 
                        child: Text(
                          'PRÓXIMOS DÍAS', 
                          style: TextStyle(color: Colors.white54, fontWeight: FontWeight.bold, fontSize: 10),
                        ),
                      ), 
                      ListView.builder(
                        shrinkWrap: true, 
                        physics: const NeverScrollableScrollPhysics(), 
                        itemCount: _forecast.length, 
                        itemBuilder: (context, index) { 
                          final day = _forecast[index]; 
                          final dayHourly = _hourly.where((h) => h['time'].toString().startsWith(day['date'])).toList(); 
                          
                          return Theme(
                            data: Theme.of(context).copyWith(dividerColor: Colors.transparent), 
                            child: ExpansionTile(
                              dense: true, 
                              iconColor: const Color(0xFF00c853), 
                              title: Row(
                                mainAxisAlignment: MainAxisAlignment.spaceBetween, 
                                children: [
                                  SizedBox(
                                    width: 80, 
                                    child: Text(
                                      _getDayName(day['date'], index), 
                                      style: TextStyle(
                                        fontSize: 15, 
                                        color: index == 0 ? const Color(0xFF00c853) : Colors.white,
                                      ),
                                    ),
                                  ), 
                                  Icon(_getWeatherIcon(day['code']), color: Colors.white70, size: 20), 
                                  Text(
                                    '${day['min']}° / ${day['max']}°', 
                                    style: const TextStyle(fontSize: 15, color: Colors.white),
                                  ),
                                ],
                              ), 
                              children: [
                                Container(
                                  height: 80, 
                                  padding: const EdgeInsets.symmetric(vertical: 4), 
                                  decoration: BoxDecoration(
                                    color: const Color(0xFF2a2a2a), 
                                    borderRadius: BorderRadius.circular(10),
                                  ), 
                                  child: ListView.builder(
                                    scrollDirection: Axis.horizontal, 
                                    itemCount: dayHourly.length, 
                                    itemBuilder: (context, hIndex) { 
                                      final hData = dayHourly[hIndex]; 
                                      return Padding(
                                        padding: const EdgeInsets.symmetric(horizontal: 12), 
                                        child: Column(
                                          mainAxisAlignment: MainAxisAlignment.center, 
                                          children: [
                                            Text(
                                              hData['time'].toString().substring(11, 16), 
                                              style: const TextStyle(color: Colors.white70, fontSize: 11),
                                            ), 
                                            const SizedBox(height: 4), 
                                            Icon(_getWeatherIcon(hData['code']), color: const Color(0xFF00c853), size: 18), 
                                            const SizedBox(height: 4), 
                                            Text(
                                              '${hData['temp']}°', 
                                              style: const TextStyle(color: Colors.white, fontWeight: FontWeight.bold, fontSize: 13),
                                            ),
                                          ],
                                        ),
                                      ); 
                                    },
                                  ),
                                ),
                              ],
                            ),
                          ); 
                        },
                      ),
                    ],
                  ),
                ),
              ),
            ],
          ),
    );
  }
}
