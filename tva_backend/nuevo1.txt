    lines_in_chunk = 0
    for w in words:
        if len(current_line) + len(w) + 1 <= limite_caracteres:
            current_line += (w + " ")
        else:
            current_chunk.append(current_line.strip())
            lines_in_chunk += 1
            current_line = w + " "
            if lines_in_chunk == 2:
                chunks.append("\\N".join(current_chunk))
                current_chunk = []
                lines_in_chunk = 0
    if current_line:
        current_chunk.append(current_line.strip())
    if current_chunk:
        chunks.append("\\N".join(current_chunk))
    return chunks

    def wrap_title_balanced(self, text):
        if not text: return "", ""
        words = text.split()
        if not words: return "", ""
        if len(words) == 1: return words[0], ""
        mid_char = sum(len(w) for w in words) // 2
        current_len = 0
        split_idx = 0
        for i, w in enumerate(words):
            current_len += len(w)
            if current_len >= mid_char:
                split_idx = i + 1
                break
        return " ".join(words[:split_idx]).strip(), " ".join(words[split_idx:]).strip()

    def tiene_audio(self, fp):
        if not fp or not os.path.exists(fp): return False
        if fp.lower().endswith(('.jpg', '.jpeg', '.png', '.webp')): return False
        try:
            res = subprocess.run(['ffprobe', '-v', 'error', '-select_streams', 'a', '-show_entries', 'stream=codec_type', '-of', 'csv=p=0', fp], capture_output=True, text=True, creationflags=0x08000000 if os.name == 'nt' else 0)
            return 'audio' in res.stdout.strip().lower()
        except: return False

            try: return float(self.dur_foto.get())
            except: return 5.0
        try:
            cmd = ['ffprobe', '-v', 'error', '-show_entries', 'format=duration', '-of', 'default=noprint_wrappers=1:nokey=1', ruta]
            res = subprocess.run(cmd, capture_output=True, text=True, creationflags=0x08000000 if os.name == 'nt' else 0)
            return float(res.stdout) if res.stdout else 0.0

    def _loop_ping(self, led_id, color_base, color_brillo, flag_name):
        if not getattr(self, flag_name, False):
            self.canvas_telemetry.itemconfig(led_id, fill=color_base)
            return
        current_fill = self.canvas_telemetry.itemcget(led_id, "fill")
        next_fill = color_brillo if current_fill == color_base else color_base
        self.canvas_telemetry.itemconfig(led_id, fill=next_fill)
        self.root.after(400, lambda: self._loop_ping(led_id, color_base, color_brillo, flag_name))

    def iniciar_ping(self, led_id, color_base, color_brillo, flag_name):
        setattr(self, flag_name, True)
        self._loop_ping(led_id, color_base, color_brillo, flag_name)

    def detener_ping(self, led_id, color_base, flag_name):

    def gestionar_subs_entrevistas(self):
        guion = self.txt_guion.get("1.0", tk.END)
        indices = list(set([int(m.group(1)) for m in re.finditer(r'\[VIDEO_AUDIO_(\d+)', guion)]))
        if not indices:
            messagebox.showinfo("Aviso", "No hay etiquetas de entrevista en el guion.")
            return
        win = ctk.CTkToplevel(self.root)
        win.title("💬 Corrector de Subtítulos de Entrevistas")
        win.geometry("700x600")
        win.grab_set()
        sf = ctk.CTkScrollableFrame(win)
        sf.pack(fill="both", expand=True, padx=10, pady=10)
        widgets_texto = {}
        target_lang = self.idioma_salida_guion.get()
        
        def transcribir(idx, tw, btn):
            btn.configure(state="disabled", text="⏳ Procesando con IA...")
            def _thread():
                try:
                    item = self.lista_media_obj[idx]
                    t_wav = os.path.join(folder_path, f"temp_ext_{idx}.wav")
                    if item["path"].lower().endswith(('.jpg', '.jpeg', '.png', '.webp')):
                        self.root.after(0, lambda: [tw.delete("1.0", tk.END), tw.insert("1.0", "[Imagen Estática - Sin Audio]"), btn.configure(state="normal", text="🤖 Auto-Transcribir")])
                        return
                    subprocess.run(['ffmpeg', '-y', '-ss', str(item.get("t_in", 0.0)), '-t', str(item.get("duration", 5.0)), '-i', item["path"], '-vn', '-af', 'afftdn=nf=-25', '-acodec', 'pcm_s16le', '-ar', '16000', '-ac', '1', t_wav], capture_output=True, creationflags=0x08000000 if os.name == 'nt' else 0)
                    mdl = self.cargar_whisper()
                    res = mdl.transcribe(t_wav, language="es", initial_prompt="Bolivia, Amazonía, Avrora Soft.")
                    txt_bruto = res["text"].strip()
                    prompt = f"ERES UNA API DE PROCESAMIENTO ESTRICTA. Arregla gramática y ortografía. DEVUELVE ÚNICA Y EXCLUSIVAMENTE EL TEXTO CORREGIDO.\n{txt_bruto}"
                    try:
                        res_o = requests.post(self.config["ollama_url"], json={"model": "dolphin-llama3", "prompt": prompt, "stream": False, "options": {"temperature": 0.0}}, timeout=120)
                        txt_final = res_o.json()["response"].strip() if res_o.ok else txt_bruto
                    except Exception as eo:
                        self.log(f"⚠️ Error Ollama: {eo}"); txt_final = txt_bruto
                    self.root.after(0, lambda: [tw.delete("1.0", tk.END), tw.insert("1.0", txt_final), btn.configure(state="normal", text="🔄 Retranscribir con IA")])
                except Exception as e:
                    self.root.after(0, lambda: [tw.insert("1.0", f"Error: {e}"), btn.configure(state="normal", text="🔄 Retranscribir con IA")])
            threading.Thread(target=_thread, daemon=True).start()

        for idx in indices:
            if idx >= len(self.lista_media_obj): continue
            f = ctk.CTkFrame(sf, fg_color="#1a1a1a")
            f.pack(fill="x", pady=5, padx=5)
            top_f = ctk.CTkFrame(f, fg_color="transparent")
            top_f.pack(fill="x", padx=5, pady=5)
            ctk.CTkLabel(top_f, text=f"Entrevista [{idx}]: {self.lista_media_obj[idx].get('nombre', 'Media')}", font=("Segoe UI", 12, "bold")).pack(side="left")
            tw = scrolledtext.ScrolledText(f, height=4, bg="#050505", fg="white", font=("Arial", 12))
            tw.pack(fill="x", padx=5, pady=5)
            if idx in self.dict_textos_entrevistas: tw.insert("1.0", self.dict_textos_entrevistas[idx])
            widgets_texto[idx] = tw
            btn_t = ctk.CTkButton(top_f, text="🤖 Auto-Transcribir", width=140, height=24, fg_color="#ffcc00", text_color="black")

    def agregar_media(self):
        archivos = filedialog.askopenfilenames(filetypes=[("Media", "*.mp4;*.mov;*.avi;*.mkv;*.jpg;*.jpeg;*.png;*.webp;*.mp3;*.wav;*.m4a;*.aac;*.ogg;*.flac;*.webm"), ("Todos", "*.*")])
        if not archivos: return

        threading.Thread(target=self._hilo_procesar_media, args=(archivos,), daemon=True).start()

    def _hilo_procesar_media(self, archivos):
        for a in archivos:
            is_vid = a.lower().endswith(('.mp4', '.mov', '.avi', '.mkv', '.webm'))
            is_img = a.lower().endswith(('.jpg', '.jpeg', '.png', '.webp'))
            dur = 5.0
            if is_vid or a.lower().endswith(('.mp3', '.wav', '.m4a', '.aac', '.ogg', '.flac')): dur = self.obtener_duracion_audio(a)
            elif is_img:
                try: dur = float(self.dur_foto.get())
                except: dur = 5.0
            if dur <= 0: dur = 5.0
            w_p, h_p = 1920, 1080
            is_vert = False
            if is_vid and HAS_CV2:
                try:
                    cap = cv2.VideoCapture(a); w_p = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH)); h_p = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT)); cap.release()
                    if h_p > w_p: is_vert = True
                except: pass
            elif is_img:
                try:
                    img = Image.open(a); w_p, h_p = img.width, img.height
                    if h_p > w_p: is_vert = True
                except: pass
            pil_thumb = self.generar_miniatura_pil(a)
            def _update_ui(path=a, is_v=is_vid, d=dur, w=w_p, h=h_p, vert=is_vert, p_t=pil_thumb):
                tk_img = ImageTk.PhotoImage(p_t)
                self.lista_media_obj.append({"path": path, "thumb": tk_img, "is_video": is_v, "duration": d, "t_in": 0.0, "v_on": True, "a_on": True, "nombre": os.path.basename(path), "transicion": "Cortar", "color": {"b": 0.0, "c": 1.0, "s": 1.0, "sh": 0}, "width": w, "height": h, "is_vid_vertical": vert})
                self.indice_media_seleccionado = len(self.lista_media_obj) - 1
                self.refrescar_galeria(); self.actualizar_monitor_estatico()
            self.root.after(0, _update_ui)
        self.root.after(0, lambda: [self.log(self.get_text("log_media_add")), self.lock_ui(False)])

    def generar_miniatura_pil(self, path):
        try:
            if HAS_CV2 and path.lower().endswith(('.mp4', '.mov', '.avi', '.mkv', '.webm')):
                cap = cv2.VideoCapture(path); s, f = cap.read(); cap.release()
                if s: img = Image.fromarray(cv2.cvtColor(f, cv2.COLOR_BGR2RGB))
                else: img = Image.new('RGB', (140, 80), (40, 40, 40))
            elif path.lower().endswith(('.mp3', '.wav', '.m4a', '.aac', '.ogg', '.flac')):
                img = Image.new('RGB', (140, 80), (20, 20, 100)); draw = ImageDraw.Draw(img); draw.text((40, 30), "AUDIO", fill="white")
            else: img = Image.open(path).convert("RGB")
            img.thumbnail((140, 80), Image.Resampling.LANCZOS); bg = Image.new('RGB', (140, 80), (0, 0, 0)); bg.paste(img, ((140 - img.width) // 2, (80 - img.height) // 2))
            return bg
        except: return Image.new('RGB', (140, 80), (50, 50, 50))

    def _construir_ui_recortador_blindado(self, item):
        import vlc
        win = ctk.CTkToplevel(self.root); win.title("🎬 Recortador Maestro"); win.geometry("950x750"); win.grab_set() 
        canvas = tk.Canvas(win, bg="#000", width=800, height=450, highlightthickness=0); canvas.pack(pady=10)
        instance = vlc.Instance("--no-xlib", "--quiet", "--no-osd"); player = instance.media_player_new()
        player.set_hwnd(canvas.winfo_id()); player.set_media(instance.media_new(item["path"]))
        dur_total = item.get("dur_original", item.get("duration", 5.0))
        var_tin, var_tout, var_pos = ctk.DoubleVar(value=item.get("t_in", 0.0)), ctk.DoubleVar(value=item.get("t_out", dur_total)), ctk.DoubleVar(value=item.get("t_in", 0.0))
        lbl_info = ctk.CTkLabel(win, text="IN: 0.00s | OUT: 0.00s", font=("Consolas", 14, "bold"), text_color=self.config["accent_blue"]); lbl_info.pack(pady=5)

        def update_ui_loop():
            if player.is_playing():
                curr_ms = player.get_time()
                if curr_ms >= 0: var_pos.set(curr_ms / 1000.0)
            win.after(100, update_ui_loop)

        def toggle_play():
            if player.is_playing(): player.pause()
            else: player.play()

        def seek_vlc(val):
            if not player.is_playing(): player.set_time(int(float(val) * 1000))

        ctk.CTkSlider(win, from_=0.0, to=dur_total, variable=var_pos, command=seek_vlc).pack(fill="x", padx=50, pady=10)
        def al(): lbl_info.configure(text=f"IN: {var_tin.get():.2f}s | OUT: {var_tout.get():.2f}s (Dur: {var_tout.get()-var_tin.get():.2f}s)")
        def si(): var_tin.set(var_pos.get()); al()
        def so(): var_tout.set(var_pos.get()); al()

        def aplicar_recorte():
            nuevo_item = item.copy()
            nuevo_item["t_in"], nuevo_item["t_out"], nuevo_item["duration"], nuevo_item["nombre"] = var_tin.get(), var_tout.get(), var_tout.get() - var_tin.get(), f"Corte_{int(time.time())}"
            self.lista_media_obj.insert(self.indice_media_seleccionado + 1, nuevo_item); self.indice_media_seleccionado += 1
            self.refrescar_galeria(); cerrar_vlc()

        def cerrar_vlc(): player.stop(); player.release(); instance.release(); win.destroy()
        win.protocol("WM_DELETE_WINDOW", cerrar_vlc)
        btn_f = ctk.CTkFrame(win, fg_color="transparent"); btn_f.pack(pady=10)
        ctk.CTkButton(btn_f, text="▶/⏸ PLAY/PAUSA", fg_color="#2e7d32", command=toggle_play).pack(side="left", padx=10)
        ctk.CTkButton(btn_f, text="[ MARCAR IN", fg_color="#ffcc00", text_color="black", command=si).pack(side="left", padx=5)
        ctk.CTkButton(btn_f, text="MARCAR OUT ]", fg_color="#ffcc00", text_color="black", command=so).pack(side="left", padx=5)
        ctk.CTkButton(win, text="✂️ CREAR SUB-CLIP", height=40, font=("Segoe UI", 12, "bold"), fg_color=self.config["accent_red"], command=aplicar_recorte).pack(pady=10)
        al(); player.play(); win.after(500, lambda: [player.pause(), player.set_time(int(var_tin.get()*1000)), update_ui_loop()])

    def toggle_play_pause(self):
        if not self.ruta_cortina_musical: return
        if self.is_playing:
            if self.is_paused: 
                pygame.mixer.music.unpause(); self.is_paused = False
                self.btn_play_pause.configure(text=self.get_text("btn_stop").replace("STOP", "PAUSA").replace("ОСТАНОВИТЬ", "ПАУЗА").replace("PARAR", "PAUSA").replace("停止", "暂停"), fg_color="#005b9f")
            else: 
                pygame.mixer.music.pause(); self.is_paused = True
                self.btn_play_pause.configure(text=self.get_text("btn_play"), fg_color="#2e7d32")
        else: self.iniciar_reproduccion_player(self.tiempo_actual_musica.get(), self.ruta_cortina_musical)

    def iniciar_reproduccion_player(self, start_pos, p_d):
        try: 
            pygame.mixer.music.load(p_d); pygame.mixer.music.set_volume(self.vol_mus_fondo.get()); pygame.mixer.music.play(0, start_pos)
            self.is_playing, self.is_paused, self.start_offset = True, False, start_pos
            self.tiempo_actual_musica.set(start_pos)
            self.btn_play_pause.configure(text=self.get_text("btn_stop").replace("STOP", "PAUSA"), fg_color="#005b9f")
            self.actualizar_timeline()
        except Exception as e: self.log(f"⚠️ Aviso: Error al iniciar reproductor: {e}")

    def detener_musica_player(self):
        try: pygame.mixer.music.stop()
        except: pass
        self.is_playing, self.is_paused = False, False
        self.tiempo_actual_musica.set(0.0)
        self.btn_play_pause.configure(text=self.get_text("btn_play"), fg_color="#2e7d32")
        if self.duracion_musica > 0: self.lbl_tiempo.configure(text=f"00:00 / {self.formatear_tiempo(self.duracion_musica)}")

    def actualizar_timeline(self):
        if self.is_playing and not self.is_paused:
            p = pygame.mixer.music.get_pos()
            if p >= 0:
                ct = self.start_offset + (p / 1000.0)
                if ct >= self.duracion_musica: self.detener_musica_player()
                else: 
                    self.tiempo_actual_musica.set(ct)
                    self.lbl_tiempo.configure(text=f"{self.formatear_tiempo(ct)} / {self.formatear_tiempo(self.duracion_musica)}")
        if self.is_playing: self.updater_id = self.root.after(100, self.actualizar_timeline)

    def on_slider_scroll(self, val):
        if self.duracion_musica > 0: self.lbl_tiempo.configure(text=f"{self.formatear_tiempo(float(val))} / {self.formatear_tiempo(self.duracion_musica)}")

    def on_slider_release(self, e):
        if self.duracion_musica > 0 and self.ruta_cortina_musical: self.iniciar_reproduccion_player(self.tiempo_actual_musica.get(), self.ruta_cortina_musical)

    def actualizar_monitor_estatico(self):
        if not HAS_CV2: return
        is_vertical = (self.formato_video.get() == "9:16")
        TW, TH = (1080, 1920) if is_vertical else (1920, 1080)
        CW, CH = (225, 400) if is_vertical else (336, 189) 
        
        self.canvas_monitor.config(width=CW, height=CH)
        self.canvas_monitor.delete("all")
        base_img = Image.new('RGB', (TW, TH), '#050505')
        
        ruta_fondo_sel = self.config["lista_fondos"].get(self.fondo_seleccionado.get(), "")
        if ruta_fondo_sel and os.path.exists(ruta_fondo_sel) and not ruta_fondo_sel.lower().endswith(('.mp4', '.mov')):
            try: base_img.paste(Image.open(ruta_fondo_sel).convert("RGB").resize((TW, TH), Image.Resampling.LANCZOS), (0, 0))
            except: pass
            
        t_lang = LANGUAGES.get(self.idioma_actual, LANGUAGES["Español"])
        opt_100p = t_lang.get("opt_100p", "100% Pantalla")
        opt_reduc = t_lang.get("opt_reduc", "Reducido Proporcional")
        
        if is_vertical:
            if self.encuadre_916.get() == opt_100p: sw_out, sh_out, vx_off, y_pip = 1080, 1920, 0, 0
            elif self.encuadre_916.get() == opt_reduc: sw_out, sh_out, vx_off, y_pip = 1080, int(1080 * 9/16), 0, (1920 - int(1080 * 9/16)) // 2
            else: sw_out, sh_out, vx_off, y_pip = 940, int(940 * 16/9), (1080 - 940) // 2, (1920 - int(940 * 16/9)) // 2
            cw = 900; cx = (1080 - cw) // 2
            h1, h2, hs, margen = 160, 160, 180, 50
            ys = 1920 - hs - margen; y2 = ys - h2; y1 = y2 - h1
        else:
            if self.encuadre_169.get() == opt_100p: sw_out, sh_out, vx_off, y_pip = 1920, 1080, 0, 0
            else: sw_out, sh_out, vx_off, y_pip = int(1920 * 0.70), int(1080 * 0.70), (1920 - int(1920 * 0.70)) // 2, (1080 - int(1080 * 0.70)) // 2 - 50 
            cw = 1400; cx = (1920 - cw) // 2
            h1, h2, hs = 80, 80, 120
            ys = 1080 - hs - 40; y2 = ys - h2; y1 = y2 - h1
        
        if self.lista_media_obj and self.indice_media_seleccionado >= 0:
            item = self.lista_media_obj[self.indice_media_seleccionado]
            if item.get("v_on", True):
                try:
                    if item["path"].lower().endswith(('.mp3', '.wav', '.m4a', '.aac', '.ogg', '.flac')):
                        p_img = Image.new('RGB', (400, 300), '#1a1a40'); draw_t = ImageDraw.Draw(p_img); draw_t.text((150, 140), "AUDIO", fill="white")
                    elif item["path"].lower().endswith(('.jpg', '.jpeg', '.png', '.webp')) or not item["path"].lower().endswith(('.mp4', '.mov', '.avi', '.mkv', '.webm')): p_img = Image.open(item["path"]).convert("RGB")
                    else:
                        cap = cv2.VideoCapture(item["path"])
                        if item.get("t_in", 0.0) > 0: cap.set(cv2.CAP_PROP_POS_MSEC, item["t_in"] * 1000)
                        suc, frame = cap.read(); cap.release()
                        if suc: p_img = Image.fromarray(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
                        else: p_img = Image.new('RGB', (400, 300), '#333333')
                    
                    is_vid_vertical = item.get("is_vid_vertical", p_img.height > p_img.width)
                    if is_vertical and self.encuadre_916.get() == opt_reduc:
                        p_img.thumbnail((sw_out, sh_out), Image.Resampling.LANCZOS)
                        p_box = Image.new('RGBA', (sw_out, sh_out), (0, 0, 0, 0)); p_box.paste(p_img, ((sw_out - p_img.width) // 2, (sh_out - p_img.height) // 2))
                    elif is_vertical and is_vid_vertical:
                        target_ratio = sw_out / sh_out; img_ratio = p_img.width / p_img.height
                        if img_ratio > target_ratio:
                            p_res = p_img.resize((int(sh_out * img_ratio), sh_out), Image.Resampling.LANCZOS)
                            left = (int(sh_out * img_ratio) - sw_out) // 2; p_box = p_res.crop((left, 0, left + sw_out, sh_out))
                        else:
                            p_res = p_img.resize((sw_out, int(sw_out / img_ratio)), Image.Resampling.LANCZOS)
                            top = (int(sw_out / img_ratio) - sh_out) // 2; p_box = p_res.crop((0, top, sw_out, top + sh_out))
                    elif is_vertical and not is_vid_vertical:
                        try:
                            gray_cv = cv2.cvtColor(np.array(p_img), cv2.COLOR_RGB2GRAY)
                            face_c = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml')
                            faces_f = face_c.detectMultiScale(gray_cv, 1.1, 4)
                            if len(faces_f) > 0:
                                fx, fy, fw, fh = faces_f[0]; scale_f = sh_out / float(p_img.height); calc_x = int((fx + fw//2) * scale_f) - (sw_out // 2)
                                crop_x_p = max(0, min(calc_x, int(p_img.width * scale_f) - sw_out))

    def limpiar_texto(self, texto, es_locucion=False):
        if not texto: return ""
        t = str(texto).replace("'", "").replace('"', "").replace(":", "").replace("{", "").replace("}", "").replace("[", "").replace("]", "")
        t = re.sub(r'[\r\n\t]+', ' ', t)
        t = re.sub(r'(?i)\b(cripta|script)\b', '', t)
        patrones = [r'^(Titular|Fecha|Noticia|Script|Locucion|Redacción|Guion|Boletín|Resumen|Audio|Texto)\s*[:\-]*\s*', r'^\s*[-*#]+\s*']
        if es_locucion:
            patrones_saludo = [r'^\s*(hola|bienvenidos|saludos|amigos|muy buenas).*?[\.,!¡]*\s*', r'^\s*buenas\s+(tardes|noches|días).*?[\.,!¡]*\s*', r'^\s*buenos\s+días.*?[\.,!¡]*\s*']
            for pat in patrones_saludo: t = re.sub(pat, '', t, flags=re.IGNORECASE).strip()
        for p in patrones: t = re.sub(p, '', t, flags=re.IGNORECASE)
        return re.sub(r'\s+', ' ', t).strip()

    def aplicar_dsp_voz(self, audio_fuente, ruta_salida):

            except: pass

        if filtros:
            f_str = ",".join(filtros)
            cmd = ['ffmpeg', '-y', '-hide_banner', '-i', audio_fuente, '-af', f_str, ruta_salida]
            try:

            v_fi, v_fo = self.safefloat(self.vid_fadein, 1.0), self.safefloat(self.vid_fadeout, 1.0)
            f_args.append(f"[{last_out_v}]fade=t=in:st=0:d={v_fi},fade=t=out:st={max(0.1, dur_total_s - v_fo)}:d={v_fo},scale=trunc(iw/2)*2:trunc(ih/2)*2,format=yuv420p[final_v]")

            amix = []
            for j, ac in enumerate(audio_chunks):
                if not self.tiene_audio(ac["path"]):
                    i_args.extend(['-f', 'lavfi', '-t', str(ac["dur"]), '-i', 'anullsrc=r=44100:cl=stereo'])
                    a_in = input_idx; input_idx += 1; delay_ms = int(ac["start"] * 1000)
                    f_args.append(f"[{a_in}:a]aformat=sample_rates=44100:channel_layouts=stereo,adelay={delay_ms}|{delay_ms}[aproc{j}]")
                    amix.append(f"[aproc{j}]")
                else:
                    i_args.extend(['-i', ac["path"]])
                    a_in = input_idx; input_idx += 1
                    t_in_val, dur_val, delay_ms, tipo = ac.get("t_in", 0.0), ac["dur"], int(ac["start"] * 1000), ac.get("tipo_audio", "broll")
                    if tipo == "voz": f_args.append(f"[{a_in}:a]atrim=start={t_in_val}:duration={dur_val},asetpts=PTS-STARTPTS,loudnorm=I=-14:TP=-1.5:LRA=11,volume={self.vol_voz.get()},aformat=sample_rates=44100:channel_layouts=stereo,adelay={delay_ms}|{delay_ms}[aproc{j}]")
                    elif tipo == "entrevista": f_args.append(f"[{a_in}:a]atrim=start={t_in_val}:duration={dur_val},asetpts=PTS-STARTPTS,loudnorm=I=-14:TP=-1.5:LRA=11,volume=1.0,aformat=sample_rates=44100:channel_layouts=stereo,adelay={delay_ms}|{delay_ms}[aproc{j}]")
                    else: f_args.append(f"[{a_in}:a]atrim=start={t_in_val}:duration={dur_val},asetpts=PTS-STARTPTS,volume=0.05,aformat=sample_rates=44100:channel_layouts=stereo,adelay={delay_ms}|{delay_ms}[aproc{j}]")
                    amix.append(f"[aproc{j}]")

            if self.ruta_cortina_musical and os.path.exists(self.ruta_cortina_musical):
                ic = input_idx
                m_in, s_climax, f_s = self.safefloat(self.mus_inicio, 0.0), self.safefloat(self.mus_salto_climax, 0.0), 2.0 
                v_i, v_f, v_c, t_fd = float(self.vol_mus_intro.get()), float(self.vol_mus_fondo.get()), float(self.vol_mus_cola.get()), self.safefloat(self.voz_retraso, 2.0)
                t_cama_muere = max(0.0, dur_total_s - d_cola); del_c = int(t_cama_muere * 1000)

                i_args.extend(['-stream_loop', '-1', '-i', self.ruta_cortina_musical]); input_idx += 1
                ix = input_idx
                i_args.extend(['-stream_loop', '-1', '-i', self.ruta_cortina_musical]); input_idx += 1
                
                f_args.append(f"[{ic}:a]atrim=start={m_in}:duration={t_cama_muere},asetpts=PTS-STARTPTS,volume='{v_i}-({v_i}-{v_f})*clip((t-{t_fd})/1.5,0,1)-{v_f}*clip((t-({t_cama_muere}-1.0))/1.0,0,1)':eval=frame,apad[a_cama]")
                
                if self.usar_climax_musica.get(): f_args.append(f"[{ix}:a]atrim=start={s_climax}:duration={d_cola},asetpts=PTS-STARTPTS,volume='{v_c}*clip(t/{f_s},0,1)':eval=frame,adelay={del_c}|{del_c},apad[a_climax]")
                else: f_args.append(f"[{ix}:a]atrim=start={m_in + t_cama_muere}:duration={d_cola},asetpts=PTS-STARTPTS,volume='{v_c}*clip(t/{f_s},0,1)':eval=frame,adelay={del_c}|{del_c},apad[a_climax]")
                
                amix.append('[a_cama]'); amix.append('[a_climax]')
                
            if len(amix) > 1: f_args.append(f"{''.join(amix)}amix=inputs={len(amix)}:duration=longest:dropout_transition=0,volume={float(len(amix))}[amix_out]")
            elif len(amix) == 1: f_args.append(f"{amix[0]}acopy[amix_out]")
            else:
                i_args.extend(['-f', 'lavfi', '-i', 'anullsrc=r=44100:cl=stereo'])
                f_args.append(f"[{input_idx}:a]anull[amix_out]"); input_idx += 1
            
            a_fi, a_fo = self.safefloat(self.aud_fadein, 1.0), self.safefloat(self.aud_fadeout, 4.0)
            f_args.append(f"[amix_out]afade=t=in:st=0:d={a_fi},afade=t=out:st={max(0.1, dur_total_s - a_fo)}:d={a_fo}[final_a]")

            fs_master_path = os.path.abspath(os.path.join(tmp_dir, "fs_master.txt"))
            with open(fs_master_path, "w", encoding="utf-8") as f: f.write(";".join(f_args))

            cmd = ['ffmpeg', '-y'] + i_args + ['-filter_complex_script', fs_master_path, '-map', '[final_v]', '-map', '[final_a]', '-c:v', 'libx264', '-preset', 'fast', '-crf', '20', '-c:a', 'aac', '-b:a', '192k', '-t', str(dur_total_s), output_video]
            res = subprocess.run(cmd, cwd=tmp_dir, capture_output=True, text=True, creationflags=0x08000000 if os.name == 'nt' else 0)
            
            if res.returncode == 0:
                self.log(f"{self.get_text('log_render_ok')} {os.path.basename(output_video)}")
                def mostrar_modal_exito():
                    dlg = ctk.CTkToplevel(self.root)
                    dlg.title("ASvideo - Sistema"); dlg.geometry("400x180"); dlg.transient(self.root); dlg.grab_set(); dlg.configure(fg_color=self.config["bg_panel"])
                    ctk.CTkLabel(dlg, text="🎬", font=("Segoe UI", 36)).pack(pady=(20, 5))
                    ctk.CTkLabel(dlg, text="PRODUCCIÓN MAESTRA COMPLETADA", font=("Segoe UI", 14, "bold"), text_color="#4ade80").pack(pady=(0, 15))
                    f_btn = ctk.CTkFrame(dlg, fg_color="transparent"); f_btn.pack()
                    def abrir_y_cerrar(): os.startfile(cfg_out); dlg.destroy()
                    ctk.CTkButton(f_btn, text="Abrir Carpeta", font=("Segoe UI", 12, "bold"), fg_color=self.config["accent_blue"], command=abrir_y_cerrar, width=120).pack(side="left", padx=10)
                    ctk.CTkButton(f_btn, text="Cerrar", font=("Segoe UI", 12), fg_color="#333333", command=dlg.destroy, width=100).pack(side="left", padx=10)
                self.root.after(0, mostrar_modal_exito)
            else: 
                err_msg = str(res.stderr)[-500:] if res.stderr else "Error desconocido o comando vacío."
                self.log(f"❌ ERROR FFmpeg: {err_msg}")

        except Exception as e:
            self.log(f"❌ ERROR GENERAL EN RENDER: {e}")
            traceback.print_exc()
        finally:
            self.root.after(0, self.progress.stop)
            self.root.after(0, lambda: [self.detener_ping(self.led_render, "#333333", "ping_active_render"), self.lock_ui(False)])
